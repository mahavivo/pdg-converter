cmake_minimum_required(VERSION 3.16)
project(pdg_converter LANGUAGES C)
if(NOT WIN32)
    message(FATAL_ERROR "This version uses Windows BCrypt and WIC; build on Windows.")
endif()
add_executable(pdg_converter pdg_converter.c)
set_target_properties(pdg_converter PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED YES)
target_compile_definitions(pdg_converter PRIVATE _WIN32_WINNT=0x0a00)
target_link_libraries(pdg_converter PRIVATE ole32 oleaut32 windowscodecs bcrypt uuid)
if(MSVC)
    target_compile_options(pdg_converter PRIVATE /O2 /W4 /utf-8)
else()
    target_compile_options(pdg_converter PRIVATE -O3 -Wall -Wextra)
    target_link_options(pdg_converter PRIVATE -municode)
endif()
